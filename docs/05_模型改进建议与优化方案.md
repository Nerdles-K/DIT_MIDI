```python
def configure_optimizers(self):
    """Configure optimizer - 使用AdamW替代Muon"""
    
    # 回到简单可靠的AdamW
    optimizer = torch.optim.AdamW(
        self.parameters(),
        lr=self.hparams.lr,
        betas=(self.hparams.lr_beta1, self.hparams.lr_beta2),
        eps=self.hparams.lr_eps,
        weight_decay=self.hparams.lr_weight_decay
    )
    
    # 保持Cosine Annealing with warmup
    def lr_lambda(step):
        if step < self.hparams.lr_warmup_steps:
            return step / self.hparams.lr_warmup_steps
        else:
            return 0.5 * (1 + np.cos(np.pi * (step - self.hparams.lr_warmup_steps) / 
                                    (self.trainer.max_steps - self.hparams.lr_warmup_steps)))
    
    scheduler = torch.optim.lr_scheduler.LambdaLR(optimizer, lr_lambda)
    
    return {
        'optimizer': optimizer,
        'lr_scheduler': {
            'scheduler': scheduler,
            'interval': 'step',
            'frequency': 1
        }
    }
```

#### OneCycle学习率策略
```python
# 在module_midi_dit.py中添加
from torch.optim.lr_scheduler import OneCycleLR

def configure_optimizers(self):
    optimizer = torch.optim.AdamW(...)
    
    # OneCycle策略: 学习率先升后降
    scheduler = OneCycleLR(
        optimizer,
        max_lr=self.hparams.lr * 3,  # 峰值3倍base lr
        total_steps=self.trainer.max_steps,
        pct_start=0.3,  # 30%时间达到峰值
        anneal_strategy='cos',
        div_factor=25.0,  # 初始lr = max_lr / 25
        final_div_factor=1000.0  # 最终lr = max_lr / 1000
    )
    
    return {
        'optimizer': optimizer,
        'lr_scheduler': {
            'scheduler': scheduler,
            'interval': 'step',
            'frequency': 1
        }
    }
```

优势:
- 动态调整学习率,前期快速收敛
- 后期精细调优
- 通常比固定余弦退火效果更好

#### 实现EMA回调
```python
# 创建 main/ema_callback.py
import torch
import pytorch_lightning as pl
from copy import deepcopy

class EMACallback(pl.Callback):
    """
    Exponential Moving Average for model weights.
    Improves generation quality and stability.
    """
    
    def __init__(self, decay=0.9999, update_every=10):
        self.decay = decay
        self.update_every = update_every
        self.ema_model = None
        
    def on_fit_start(self, trainer, pl_module):
        self.ema_model = deepcopy(pl_module.model)
        self.ema_model.eval()
        self.ema_model.requires_grad_(False)
        
    def on_train_batch_end(self, trainer, pl_module, outputs, batch, batch_idx):
        if batch_idx % self.update_every == 0:
            self._update_ema(pl_module.model)
            
    def _update_ema(self, model):
        with torch.no_grad():
            for ema_param, model_param in zip(
                self.ema_model.parameters(), 
                model.parameters()
            ):
                ema_param.data.mul_(self.decay).add_(
                    model_param.data, alpha=1 - self.decay
                )
    
    def on_validation_start(self, trainer, pl_module):
        # Use EMA model for validation
        self.original_model = pl_module.model
        pl_module.model = self.ema_model
        
    def on_validation_end(self, trainer, pl_module):
        # Restore original model
        pl_module.model = self.original_model
```

class MIDIQualityFilter:
    """过滤低质量MIDI文件"""
    
    def is_valid(self, midi_path):
        features = extract_features(midi_path)
        
        # 过滤条件
        if features['duration'] < 5.0:  # 太短
            return False
        if features['note_density'] < 1.0:  # 太稀疏
            return False
        if len(features['instruments']) == 0:  # 无乐器
